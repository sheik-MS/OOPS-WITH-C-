INLINE FUNCTION FormatPrefix(prefixTag: STRING) RETURNS STRING
    RETURN "[" + prefixTag + "] "
END FUNCTION
FUNCTION LogEvent(message: STRING)
    PRINT FormatPrefix("INFO") + message
END FUNCTION
FUNCTION LogEvent(message: STRING, severityLevel: INTEGER, timestamp: STRING = "00:00:00")
    severity: STRING
    IF severityLevel = 1 THEN severity <- "LOW"
    ELSE IF severityLevel = 2 THEN severity <- "MEDIUM"
    ELSE IF severityLevel = 3 THEN severity <- "CRITICAL"
    ELSE severity <- "UNKNOWN"
    ENDIF
    PRINT FormatPrefix(severity) + "@ " + timestamp + " : " + message
END FUNCTION
FUNCTION LogEvent(errorCode: INTEGER, systemModule: STRING)
    PRINT FormatPrefix("HARDWARE_FAULT") + "Module: " + systemModule
          + " -> Error Code: 0x" + TO_HEX(errorCode)
END FUNCTION
BEGIN MAIN
    LogEvent("System started")
    LogEvent("Motor warm", 2)
    LogEvent("Motor stopped", 3, "12:30:00")
    LogEvent(26, "MOTOR")
END MAIN
