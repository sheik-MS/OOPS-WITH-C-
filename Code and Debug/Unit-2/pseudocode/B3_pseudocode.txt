CLASS StudentRecord
PRIVATE:
    rollNo: INTEGER
    studentName: STRING
    marks: ARRAY[3] OF REAL
    percentage: REAL
    grade: CHARACTER
    METHOD CalculateResult()
        percentage <- (marks[0] + marks[1] + marks[2]) / 3
        IF percentage >= 85 THEN grade <- 'A'
        ELSE IF percentage >= 70 THEN grade <- 'B'
        ELSE IF percentage >= 50 THEN grade <- 'C'
        ELSE grade <- 'F'
        ENDIF
    END METHOD
PUBLIC:
    CONSTRUCTOR()
        rollNo <- 0; studentName <- ""; percentage <- 0; grade <- 'F'
        FOR i <- 0 TO 2 DO marks[i] <- 0 ENDFOR
    END CONSTRUCTOR
    METHOD ReadData(r: INTEGER, name: STRING, m1: REAL, m2: REAL, m3: REAL)
        rollNo <- r; studentName <- name
        marks[0] <- m1; marks[1] <- m2; marks[2] <- m3
        CalculateResult()
    END METHOD
    METHOD GetPercentage() RETURNS REAL
        RETURN percentage
    END METHOD
    METHOD Display()
        PRINT rollNo, studentName, percentage, grade
    END METHOD
ENDCLASS
BEGIN MAIN
    roster: ARRAY[3] OF StudentRecord
    roster[0].ReadData(101,"Aarav",60,70,80)
    roster[1].ReadData(102,"Bhavna",90,90,90)
    roster[2].ReadData(103,"Chirag",75,75,75)
    FOR pass <- 0 TO 1
        FOR j <- 0 TO 1 - pass
            IF roster[j].GetPercentage() < roster[j+1].GetPercentage() THEN
                SWAP roster[j], roster[j+1]
            ENDIF
        ENDFOR
    ENDFOR
    FOR i <- 0 TO 2 DO roster[i].Display() ENDFOR
END MAIN
